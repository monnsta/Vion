## Getting Started
The application is designed to run robustly with built-in fail-safes. There are two primary methods for starting the bot depending on your environment (Production vs. Development).

### Production Environment (Recommended)
Execute the provided bash script. This acts as a lightweight process manager that installs dependencies, provides security over the process, automatically restarts the bot in the event of a crash, and detects crash loops to prevent system overload.
```bash
./exec.sh
```

### Development Environment
For testing or debugging, you can bypass the bash script and run the bot script directly.
```bash
python bot.py
```

---

## Configuration
The bot's identity, behavior, limits, and AI parameters are entirely driven by the configure.yml file located in the project root.

## Project Architecture
The repository is modularized to separate core logic, configuration, and runtime execution smoothly.
```

│   ├── cog/                # Discord commands and events extensions

### Directory Breakdown
 * **core/**: The brain of the bot. This holds all executable logic.
   * **cog/**: The command and event handlers loaded into the bot instance.
   * **lib/**: Foundational utilities handling file IO, isolated code execution, and data formatting.
   * **ui/**: Pre-built interface structures (like embeds or views) used across commands.
   * **wrp/**: Simple APIs built over complex systems (wrappers). This allows the cogs to easily interact with memory, data, and blacklists without writing boilerplate.
 * **meta/**: Handles the ingestion of environmental variables and parsing configure.yml upon initialization.
 * **Root Scripts (bot.py, bootstrap.py, exec.sh)**: The structural layers that boot up the logging environment, establish the Discord connection, and manage the operating system process.
```

import bootstrap

import asyncio
import logging
import signal

import discord
from discord.ext         import commands

from core.lib.bttrdict   import BD, Repr
from core.wrp.profile    import server, user
from core.wrp.extension  import extension
from meta                import config


log = logging.getLogger(__name__)


class Bot(commands.Bot):
	def __init__(self):
		intents = discord.Intents.default()
		intents.members = True
		intents.presences = True
		intents.message_content = True

		super().__init__(
			command_prefix=self.determine_prefix,
			intents=intents
		)
		log.info("Bot initialized")

	@staticmethod
	def _map_command_mentions(synced_commands):
		def walk(command, prefix="", root_id=None):
			if root_id is None:
				root_id = command.id

			name = f"{prefix} {command.name}".strip()
			var_name = name.replace(" ", ".")
			mention_name = f"</{name}:{root_id}>"

			if isinstance(command, discord.app_commands.Command):
				command_data = BD(
					name        = name,
					mention     = mention_name,
					description = command.description or "No description.",
					root_id     = root_id,
					parameters  = command.parameters
				)

				config.cmd.set(
					var_name,
					Repr(mention_name, command_data)
				)

			elif isinstance(command, discord.app_commands.Group):
				for sub_item in command.commands:
					walk(sub_item, prefix=name, root_id=root_id)

		for cmd in synced_commands:
			walk(cmd)

	async def setup_hook(self):
		log.info("Starting Setup...")

		loop = asyncio.get_running_loop()
		for sig in (signal.SIGINT, signal.SIGTERM):
			try:
				loop.add_signal_handler(sig, lambda: asyncio.create_task(self.shutdown()))
			except NotImplementedError:
				pass

		dev_guilds = [discord.Object(id=g[0]) for g in config.dev_guilds]

		await extension.load_all(self)

		log.info("Syncing application commands...")
		synced_global = await self.tree.sync()

		total_dev_synced = 0
		for guild in dev_guilds:
			dev_synced = await self.tree.sync(guild=guild)
			total_dev_synced += len(dev_synced)

		log.info(f"Sync Complete: {len(synced_global)} Global, {total_dev_synced} Dev")

		self._map_command_mentions(synced_global)

	async def determine_prefix(self, _, message):
		_, guild_settings = await server.settings.load(message.guild.id)
		_, user_settings  = await user.settings.load(message.author.id) 

		return [
			*config.prefix,
			*guild_settings.get("prefix", []),
			*user_settings.get("prefix", [])
		]


if __name__ == "__main__":
	bot = Bot()

	try:
		log.info("Starting bot...")
		bot.run(config.secrets.discord_api_key)
	except (KeyboardInterrupt, SystemExit):
		log.info("Process terminated by user")
	except Exception:
		log.critical("Fatal error during bot startup", exc_info=True)

exec.sh:
```sh
#!/bin/bash

# --- 1. Configuration ---
PYTHON_BIN="python3"
PIP_BIN="pip3"
BOT_FILE="bot.py"
REQUIREMENTS="requirements.txt"

RESTART_WAIT=2			# Seconds to wait before restarting
MAX_CRASHES=2			# Max failures allowed...
CRASH_WINDOW=60			# ...within this many seconds


# --- 2. Internal Setup ---
APP_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" &> /dev/null && pwd)
BOT_PATH="$APP_DIR/$BOT_FILE"
REQ_PATH="$APP_DIR/$REQUIREMENTS"
CRASH_TS=()

# Colors for terminal output
RED='\e[1;31m'
GRN='\e[1;32m'
YLW='\e[1;33m'
CYN='\e[1;36m'
RES='\e[0m'


# --- 3. Functions ---
log() {
	echo -e "${CYN}[$(date '%H:%M:%S')]${RES} ${GRN}[INFO   ]${RES} exec.sh: $1"
}

warn() {
	echo -e "${CYN}[$(date '%H:%M:%S')]${RES} ${YLW}[WARNING]${RES} exec.sh: $1"
}

error() {
	echo -e "${CYN}[$(date '%H:%M:%S')]${RES} ${RED}[ERROR  ]${RES} exec.sh: $1" >&2
}

check_env() {
	if ! command -v "$PYTHON_BIN" &> /dev/null; then
		error "Python not found: $PYTHON_BIN"
		exit 1
	fi

	if ! command -v "$PIP_BIN" &> /dev/null; then
		error "Pip not found: $PIP_BIN"
		exit 1
	fi

	if [[ ! -f "$BOT_PATH" ]]; then
		error "Bot script not found: $BOT_PATH"
		exit 1
	fi
}

install_deps() {
	if [[ -f "$REQ_PATH" ]]; then
		log "Checking dependencies..."
		if ! "$PIP_BIN" install -r "$REQ_PATH" --quiet; then
			warn "Dependency installation failed. Attempting to run anyway."
		fi
	fi
}

is_crash_looping() {
	local now=$(date +%s)
	local cutoff=$((now - CRASH_WINDOW))
	
	local updated_ts=()
	for ts in "${CRASH_TS[@]}"; do
		[[ "$ts" -gt "$cutoff" ]] && updated_ts+=("$ts")
	done
	
	updated_ts+=("$now")
	CRASH_TS=("${updated_ts[@]}")

	if [[ ${#CRASH_TS[@]} -ge $MAX_CRASHES ]]; then
		return 0 # True
	fi
	return 1 # False
}

run_docker() {
	if ! command -v docker &> /dev/null; then
		error "Docker not found. Please install Docker first."
		exit 1
	fi
	
	log "Building and starting containers (detached)..."
	docker compose up --build -d
	log "Container is up. Use 'docker compose logs -f' to view output."
	exit 0
}


# --- 4. Main Execution ---
trap "log 'Manual interruption detected. Exiting...'; exit 0" SIGINT SIGTERM

if [[ "$1" == "--docker" ]]; then
	run_docker
fi

check_env
install_deps

log "Supervisor active. Monitoring $BOT_FILE..."

while true; do
	"$PYTHON_BIN" "$BOT_PATH"
	EXIT_CODE=$?

	case $EXIT_CODE in
		0)
			log "Clean exit detected. Stopping."
			break
			;;
		10)
			log "Critical error. Exiting."
			break
			;;
		42)
			log "System requested immediate restart..."
			sleep 1
			;;
		130 | 143)
			log "Termination signal ($EXIT_CODE). Exiting."
			break
			;;
		*)
			error "Bot crashed with code $EXIT_CODE."
			
			if is_crash_looping; then
				error "FATAL: Reached $MAX_CRASHES crashes in ${CRASH_WINDOW}s. Manual intervention required."
				exit 1
			fi
			
			log "Restarting in ${RESTART_WAIT}s (Failure ${#CRASH_TS[@]}/$MAX_CRASHES)..."
			sleep "$RESTART_WAIT"
			;;
	esac
done
```

import logging
from datetime       import datetime, timedelta
from typing         import Any, Dict, List, Optional, Tuple

from core.lib.file  import json_file
from meta           import config, file



log = logging.getLogger(__name__)


class Blacklist:
	@staticmethod
	def _now() -> datetime:
		return datetime.now(config.timezone)

	def _parse_duration(self, duration: str) -> Optional[str]:
		try:
			amount = int(duration[:-1])
			unit = duration[-1].lower()

			if unit == "d":
				delta = timedelta(days=amount)
			elif unit == "h":
				delta = timedelta(hours=amount)
			elif unit == "m":
				delta = timedelta(minutes=amount)
			else:
				log.debug(f"Invalid duration unit: {unit}")
				return None

			expiry = (self._now() + delta).isoformat()
			log.debug(f"Parsed duration {duration} to expire at {expiry}")
			return expiry
		except Exception as e:
			log.error(f"Failed to parse duration '{duration}': {e}")
			return None

	async def _get_data(self) -> Tuple[bool, Dict[str, Dict[str, Any]]]:
		success, data = await json_file.load(file.glob.blacklist)

		if "server" not in data: data["server"] = {}
		if "user" not in data: data["user"] = {}

		if not success and not data:
			return False, data

		now = self._now()
		modified = False
		expired_count = 0

		for e_type in ("server", "user"):
			for e_id in list(data[e_type].keys()):
				entry = data[e_type][e_id]
				if entry.get("expires_at"):
					expiry = datetime.fromisoformat(entry["expires_at"])
					if expiry <= now:
						del data[e_type][e_id]
						expired_count += 1
						modified = True

		if modified:
			await json_file.save(file.glob.blacklist, data)

		return True, data

	async def add(self, entity_id: int, entity_type: str, mod_id: int, reason: str, duration: Optional[str] = None) -> Tuple[bool, str]:
		if entity_type not in ("server", "user"):
			log.warning(f"Attempted to blacklist invalid entity type: {entity_type}")
			return False, "Invalid entity type"

		success, data = await self._get_data()
		if not success:
			return False, "Failed to load blacklist database."

		eid = str(entity_id)
		if eid in data[entity_type]:
			log.info(f"Blacklist add failed: {entity_type} {eid} is already blacklisted")
			return False, "Entity is already blacklisted."

		expires_at = self._parse_duration(duration) if duration else None
		if duration and not expires_at:
			return False, "Invalid duration format (use 7d, 12h, 30m)."

		data[entity_type][eid] = {
			"id": entity_id,
			"type": entity_type,
			"mod_id": mod_id,
			"reason": reason,
			"created_at": self._now().isoformat(),
			"expires_at": expires_at,
		}

		await json_file.save(file.glob.blacklist, data)
		log.info(f"Successfully blacklisted {entity_type} {eid} by mod {mod_id}. Reason: {reason}")
		return True, "Blacklisted successfully."

	async def remove(self, entity_id: int, entity_type: str) -> Tuple[bool, str]:
		if entity_type not in ("server", "user"):
			return False, "Invalid entity type"

		success, data = await self._get_data()
		if not success:
			return False, "Failed to load blacklist database."

		eid = str(entity_id)

		if eid in data[entity_type]:
			del data[entity_type][eid]
			await json_file.save(file.glob.blacklist, data)
			log.info(f"Successfully removed {entity_type} {eid} from blacklist")
			return True, "Removed from blacklist."

		log.info(f"Blacklist remove failed: {entity_type} {eid} not found")
		return False, f"{entity_type.capitalize()} was not blacklisted."

	async def get(self, entity_id: int, entity_type: str) -> Tuple[bool, Any]:
		log.debug(f"Fetching blacklist info for {entity_type} {entity_id}")

		if entity_type not in ("server", "user"):
			return False, "Invalid entity type"

		success, data = await self._get_data()
		if not success:
			return False, "OS error"

		eid = str(entity_id)
		if eid in data[entity_type]:
			return True, data[entity_type][eid]

		return False, f"{entity_type.capitalize()} with id '{eid}' was not found."

	async def list_all(self, entity_type: Optional[str] = None) -> List[Dict[str, Any]]:
		log.debug(f"Listing all blacklisted entities (filter: {entity_type})")
		success, data = await self._get_data()

		if not success:
			return []

		if entity_type:
			return list(data.get(entity_type, {}).values())

		return list(data["server"].values()) + list(data["user"].values())

	async def is_blacklisted(self, entity_id: int, entity_type: str) -> bool:
		if entity_type not in ("server", "user"):
			return False

		success, data = await self._get_data()
		if not success:
			return False

		result = str(entity_id) in data[entity_type]
		log.debug(f"Check is_blacklisted for {entity_type} {entity_id}: {result}")
		return result

class _EntityListStore:
	def __init__(self, key: str, max_items: int, entity_type: str = "guild"):
		self.key = key
		self.max_items = max_items
		self.entity_type = entity_type

	def _path(self, entity_id: int) -> str:
		eid = str(entity_id)
		if self.entity_type == "guild":
			return file.server.ai_data(eid)
		elif self.entity_type == "user":
			return file.user.ai_data(eid)
		else:
			log.error(f"Unknown entity type encountered: {self.entity_type}")
			raise ValueError(f"Unknown entity type: {self.entity_type}")

	async def _load(self, entity_id: int) -> tuple[bool, dict]:
		log.debug(f"Loading {self.key} for {self.entity_type} {entity_id}")
		return await json_file.load(self._path(entity_id))

	async def _save(self, entity_id: int, data: dict) -> None:
		log.debug(f"Saving {self.key} for {self.entity_type} {entity_id}")
		await json_file.save(self._path(entity_id), data)

	def _get_list(self, data: dict) -> list[Any]:
		return data.setdefault(self.key, [])

	def _trim(self, items: list[Any]) -> list[Any]:
		if len(items) > self.max_items:
			log.debug(f"Trimming {self.key} list: {len(items)} -> {self.max_items}")
			return items[-self.max_items:]
		return items

	async def get(self, entity_id: int) -> tuple[bool, list[Any]]:
		try:
			success, data = await self._load(entity_id)
			if not success:
				log.warning(f"Failed to load data for {self.entity_type} {entity_id} while getting {self.key}")
				return False, []
			return True, data.get(self.key, [])
		except Exception as e:
			log.error(f"Error retrieving {self.key} for {self.entity_type} {entity_id}: {e}", exc_info=True)
			return False, []

	async def clear(self, entity_id: int) -> bool:
		try:
			success, data = await self._load(entity_id)
			if not success:
				return False
			data[self.key] = []
			await self._save(entity_id, data)
			log.info(f"Cleared {self.key} for {self.entity_type} {entity_id}")
			return True
		except Exception as e:
			log.error(f"Error clearing {self.key} for {self.entity_type} {entity_id}: {e}")
			return False

	class Data:

		async def exists(self, entity_id: int) -> bool:
			eid = str(entity_id)
			path = Path(self.parent.active_dir) / eid

			return await asyncio.to_thread(path.exists)

		async def delete(self, entity_id: int) -> bool:

				return False

			binned = Path(sub.trash) / eid
			active = Path(self.parent.active_dir) / eid

			if not binned.exists():
				log.warning(f"Restore failed: Binned data for {eid} not found")
				return False

			temp_backup = active.with_suffix(".bak")

						info_file.unlink()

				log.info(f"Successfully restored {self.parent.entity_type} {eid}")
				return True
			except Exception:
				log.error(f"Restore failed for '{eid}'", exc_info=True)
				def rollback():
					if temp_backup.exists():
						log.info(f"Rolling back restoration for {eid}")
						temp_backup.rename(active)
				await asyncio.to_thread(rollback)
				return False

		async def delete_permanent(self, entity_id: int) -> tuple[bool, str]:
			eid = str(entity_id)
			source = Path(self.parent.active_dir) / eid
			binned = Path(sub.trash) / eid

			def sync_delete_logic():
				path = binned if binned.exists() else source if source.exists() else None

				if not path:
					log.warning(f"Permanent delete failed: {eid} not found in active or bin")
					return False, "Could not find any data"

				log.info(f"Permanently purging {self.parent.entity_type} {eid}")
				shutil.rmtree(path)

			try:
				return await asyncio.to_thread(sync_delete_logic)
			except Exception:
				log.error(f"Error during permanent delete for {eid}", exc_info=True)
				return False, "There was an error deleting your data, please report this"

		async def get_zip(self, entity_id: int) -> tuple[bool, Union[io.BytesIO, str]]:
			eid = str(entity_id)
			source_dir = Path(self.parent.active_dir) / eid

			if not source_dir.exists():
				log.warning(f"Zip failed: Source path {source_dir} does not exist")
				return False, "We could not find any data related to you in our storage."

			def create_in_memory_zip():
				buffer = io.BytesIO()
				try:
					def dict_to_markdown(data_obj, level=0) -> str:
						md = ""
						indent = "\t" * level
						if isinstance(data_obj, dict):
							for key, value in data_obj.items():
								if isinstance(value, (dict, list)) and value:
									md += f"{indent}- **{key}**:\n"
									md += dict_to_markdown(value, level + 1)
								else:
									md += f"{indent}- **{key}**: {value}\n"
						elif isinstance(data_obj, list):
							for item in data_obj:
								if isinstance(item, (dict, list)):
									md += dict_to_markdown(item, level + 1)
								else:
									md += f"{indent}- {item}\n"
						else:
							md += f"{indent}{data_obj}\n"
						return md

					with zipfile.ZipFile(buffer, "w", zipfile.ZIP_DEFLATED) as zf:
						readme_content = (
							f"# {self.parent.entity_type.capitalize()} Data Export\n\n"
							"This archive contains a readable export of your requested data.\n\n"
							"### File Guide:\n"
							"- **settings.md**: Your core preferences and configuration settings.\n"
							"- **ai_data.md**: Cached information the AI uses for processing.\n"
						)
						if self.parent.entity_type == "guild":
							readme_content += (
								"- **commands.py**: Custom commands setup for this server.\n"
								"- **events.py**: Custom events setup for this server.\n"
							)
						
						readme_content += "\n*Note: Original JSON data files have been formatted as readable Markdown lists for easier viewing.*"
						zf.writestr("README.md", readme_content)

						for file_path in source_dir.rglob('*'):
							if not file_path.is_file():
								continue

							rel_path = file_path.relative_to(source_dir)

							if file_path.suffix == '.json':
								try:
									with file_path.open('r', encoding='utf-8') as f:
										json_data = json.load(f)

									md_content = f"# {file_path.stem.capitalize()} Data\n\n"
									md_content += dict_to_markdown(json_data)

									md_rel_path = rel_path.with_suffix(".md")
									zf.writestr(str(md_rel_path).replace('\\', '/'), md_content)
								except json.JSONDecodeError:
									zf.write(file_path, str(rel_path).replace('\\', '/'))
							else:
								zf.write(file_path, str(rel_path).replace('\\', '/'))

					buffer.seek(0)
					return True, buffer
				except Exception:
					log.error(f"Error creating zip for {eid}", exc_info=True)
					return False, "There was an error gathering your data. Please report this."

			log.info(f"Generating archive for {self.parent.entity_type} {eid}")
			return await asyncio.to_thread(create_in_memory_zip)

class Extension:
	@staticmethod
	def _get_cog_name(extension: str) -> str:
		return extension.split('.')[-1].capitalize()

	async def _handle_dev_guilds(self, bot: commands.Bot, extension: str, dev_guilds: Iterable | None = None) -> None:
		if not dev_guilds:
			return

		cog_name = self._get_cog_name(extension)
		cog = bot.get_cog(cog_name)

		if cog and getattr(cog, 'dev_only', False):
			for command in cog.get_app_commands():
				for guild in dev_guilds:
					bot.tree.add_command(command, guild=guild, override=True)
					bot.tree.remove_command(command.name)
			log.debug(f"Registered app commands for {extension} in dev guilds.")

	async def get_all(self) -> List[str]:
		extensions = []
		cog_dir = Path(sub.cog)

		for path in cog_dir.rglob('*.py'):
			if any(part.startswith('_') for part in path.parts):
				continue

			extension_path = str(path.with_suffix('')).replace('\\', '.').replace('/', '.')
			extensions.append(extension_path)

		return extensions

	async def add(self, bot: commands.Bot, guild_id: int | str, code: str) -> Tuple[bool, str]:
		guild_id = str(guild_id)
		cog_path = file.server.commands(guild_id)

		module_path = str(cog_path.with_suffix('')).replace('\\', '.').replace('/', '.')

		try:
			await create_path((cog_path, False))

			await text_file.save(str(cog_path), code)

			if module_path in bot.extensions:
				return await self.reload(bot, module_path)
			return await self.load(bot, module_path)

		except SyntaxError as e:
			log.error(f"Syntax error in custom cog for guild {guild_id}: {e}")
			return False, f"Failed to load due to Syntax Error:\n```py\n{e}\n```"
		except Exception as e:
			log.error(f"Failed to create/load custom cog for {guild_id}", exc_info=True)
			return False, f"Critical error during custom cog deployment:\n```py\n{e}\n```"

	async def load(self, bot: commands.Bot, extension: str) -> Tuple[bool, str]:
		try:
			await bot.load_extension(extension)
			await self._handle_dev_guilds(bot, extension, config.dev_server_ids)
			log.info(f"Loaded cog: '{extension}'")
			return True, f"Successfully loaded `{extension}`."
		except commands.ExtensionAlreadyLoaded:
			return False, f"`{extension}` is already loaded."
		except commands.NoEntryPointError:
			return False, f"`{extension}` is missing a `setup` function."
		except commands.ExtensionFailed as e:
			return False, f"Extension failed to initialize:\n```py\n{e.original}\n```"
		except Exception as e:
			log.error(f"Failed to load {extension}", exc_info=True)
			return False, f"Failed to load:\n```py\n{e}\n```"

	async def unload(self, bot: commands.Bot, extension: str) -> Tuple[bool, str]:
		if "dev" in extension:
			return False, "This cog cannot be unloaded."

		try:
			await bot.unload_extension(extension)
			log.info(f"Unloaded cog: '{extension}'")
			return True, f"Successfully unloaded `{extension}`."
		except commands.ExtensionNotLoaded:
			return False, f"`{extension}` is not loaded."
		except Exception as e:
			log.error(f"Failed to unload {extension}", exc_info=True)
			return False, f"Failed to unload:\n```py\n{e}\n```"

	async def reload(self, bot: commands.Bot, extension: str) -> Tuple[bool, str]:
		try:
			await bot.reload_extension(extension)
			await self._handle_dev_guilds(bot, extension, config.dev_server_ids)
			log.info(f"Reloaded cog: '{extension}'")
			return True, f"Successfully reloaded `{extension}`."
		except commands.ExtensionNotLoaded:
			return await self.load(bot, extension)
		except Exception as e:
			log.error(f"Failed to reload {extension}", exc_info=True)
			return False, f"Failed to reload:\n```py\n{e}\n```"

	async def load_all(self, bot: commands.Bot) -> Dict[str, Tuple[bool, str]]:
		results = {}
		for ext in await self.get_all():
			results[ext] = await self.load(bot, ext)
		return results

	async def unload_all(self, bot: commands.Bot) -> Dict[str, Tuple[bool, str]]:
		results = {}
		loaded_exts = list(bot.extensions.keys())

		for ext in loaded_exts:
			if "cog_manager" not in ext:
				results[ext] = await self.unload(bot, ext)
		return results

	async def reload_all(self, bot: commands.Bot) -> Dict[str, Tuple[bool, str]]:
		results = {}
		loaded_exts = list(bot.extensions.keys())

		for ext in loaded_exts:
			results[ext] = await self.reload(bot, ext)
		return results

import json

from typing       import Any, Callable, Dict, List, Optional, Set, Union, Type, TypeVar, Iterator

T = TypeVar("T", bound="BD")


# Error Classes
class DataError(Exception):
	"""Base exception for BD data operations."""
	pass

class ComputationError(DataError):
	"""Raised when a Computed or Lazy value fails to evaluate."""
	pass

class ImmutableError(DataError):
	"""Raised when attempting to modify a frozen BD object."""
	pass


# Wrappers
class Repr:
	"""
	A proxy wrapper that returns a specific string representation
	while delegating all attribute access and comparisons to an 
	underlying value.
	"""
	__slots__ = ("_repr_val", "_actual_val")

	def __init__(self, repr_val: str, actual_val: Any):
		self._repr_val = repr_val
		self._actual_val = actual_val

	def __getattr__(self, key: str) -> Any:
		return getattr(self._actual_val, key)

	def __getitem__(self, key: Any) -> Any:
		return self._actual_val[key]

	# Comparison Operators
	def __lt__(self, other: Any) -> bool:
		return self._actual_val < (other._actual_val if isinstance(other, Repr) else other)

	def __le__(self, other: Any) -> bool:
		return self._actual_val <= (other._actual_val if isinstance(other, Repr) else other)

	def __eq__(self, other: Any) -> bool:
		return self._actual_val == (other._actual_val if isinstance(other, Repr) else other)

	def __gt__(self, other: Any) -> bool:
		return self._actual_val > (other._actual_val if isinstance(other, Repr) else other)

	def __ge__(self, other: Any) -> bool:
		return self._actual_val >= (other._actual_val if isinstance(other, Repr) else other)

	def __str__(self) -> str:
		return self._repr_val

	def __repr__(self) -> str:
		return self._repr_val

class AntiFreeze:
	"""Wrapper to protect specific keys from being frozen."""
	__slots__ = ("value",)
	def __init__(self, value: Any):
		self.value = value

class Method:
	"""Wrapper to mark a function as an instance method."""
	__slots__ = ("fn",)
	def __init__(self, fn: Callable):
		if not callable(fn):
			raise TypeError("Method expects a callable function.")
		self.fn = fn

class Computed:
	"""Wrapper for values calculated dynamically from the BD state."""
	__slots__ = ("fn", "lazy")
	def __init__(self, fn: Callable[["BD"], Any], lazy: bool = False):
		if not callable(fn):
			raise TypeError("Computed expects a callable function.")
		self.fn = fn
		self.lazy = lazy

# Core Data Object
class BD:
	"""
	BttrDict (BD): A nested, observable, and freezable data container
	supporting dot-notation, computed properties, and custom representations.
	"""
	__slots__ = ("_data", "_watchers", "_frozen", "_anti_freeze")

	def __init__(self, **kwargs: Any):
		super().__setattr__("_data", {})
		super().__setattr__("_watchers", [])
		super().__setattr__("_frozen", False)
		super().__setattr__("_anti_freeze", set())

		for k, v in kwargs.items():
			self.set(k, v)

	@classmethod
	def load(cls: Type[T], data: Union[str, Dict[str, Any]]) -> T:
		"""Loads a dictionary or JSON string, recursively converting nested dicts to BD instances."""
		if isinstance(data, str):
			try:
				parsed_data = json.loads(data)
			except json.JSONDecodeError as e:
				raise DataError(f"Failed to parse JSON string: {e}")
		elif isinstance(data, dict):
			parsed_data = data
		else:
			raise TypeError("load() expects a JSON string or a dictionary.")

		if not isinstance(parsed_data, dict):
			raise DataError("Root structure must be a dictionary/object.")

		def _convert(item: Any) -> Any:
			if isinstance(item, dict):
				return cls(**{k: _convert(v) for k, v in item.items()})
			if isinstance(item, list):
				return [_convert(i) for i in item]
			if isinstance(item, set):
				return {_convert(i) for i in item}
			if isinstance(item, tuple):
				return tuple(_convert(i) for i in item)
			return item

		return _convert(parsed_data)

	def __repr__(self) -> str:
		def _format_val(val: Any) -> str:
			if isinstance(val, (BD, Repr)):
				return repr(val)
			if isinstance(val, (dict, MappingProxyType)):
				inner = ", ".join(f"{repr(k)}: {_format_val(v)}" for k, v in val.items())
				return "{" + inner + "}"
			if isinstance(val, (list, tuple, set, frozenset)):
				brackets = {list: "[]", tuple: "()", set: "{}", frozenset: "{}"}
				b = brackets.get(type(val), "[]")
				inner = ", ".join(_format_val(i) for i in val)
				return f"{b[0]}{inner}{b[1]}"
			if type(val) is Computed:
				return "<computed>"
			return repr(val)

		items = [f"{k}={_format_val(v)}" for k, v in self._data.items() 
				 if not isinstance(v, types.MethodType)]
		return f"<{', '.join(items)}>"

	def __len__(self) -> int:
		return len(self._data)

	def __iter__(self) -> Iterator[str]:
		return iter(self._data)

	def __contains__(self, key: str) -> bool:
		return key in self._data

	def __eq__(self, other: Any) -> bool:
		if isinstance(other, BD):
			return self.to_dict() == other.to_dict()
		if isinstance(other, dict):
			return self.to_dict() == other
		return False

	def __getitem__(self, path: str) -> Any:
		val = self.get(path, default=KeyError)
		if val is KeyError:
			raise KeyError(f"Path '{path}' not found.")
		return val

	def __setitem__(self, path: str, value: Any) -> None:
		self.set(path, value)

	def __delitem__(self, path: str) -> None:
		self.delete(path)

	def __getattr__(self, key: str) -> Any:
		try:
			val = self._data[key]
		except KeyError:
			raise AttributeError(f"'{type(self).__name__}' object has no attribute '{key}'")

		if type(val) is Computed:
			try:
				result = val.fn(self)
				self._data[key] = result
				return result
			except Exception as e:
				raise ComputationError(f"Lazy computation failed for '{key}': {e}") from e

		if type(val) is Alias:
			return self.get(val.path)

		if type(val) is Secret:
			return val.value

		return val

	def __setattr__(self, key: str, value: Any) -> None:
		if key in type(self).__slots__:
			super().__setattr__(key, value)
			return
		self.set(key, value)

	def keys(self) -> Iterator[str]:
		return self._data.keys()

	def values(self) -> Iterator[Any]:
		return (getattr(self, k) for k in self._data)

	def items(self) -> Iterator[tuple[str, Any]]:
		return ((k, getattr(self, k)) for k in self._data)

	def update(self, data: Union[Dict[str, Any], "BD"]) -> "BD":
		"""Updates the BD instance with another dictionary or BD instance."""
		iterable = data._data.items() if isinstance(data, BD) else data.items()
		for k, v in iterable:
			self.set(k, v)
		return self

	def pop(self, key: str, default: Any = KeyError) -> Any:
		"""Removes a key and returns its value."""
		if key in self._data:
			val = getattr(self, key)
			self.delete(key)
			return val
		if default is not KeyError:
			return default
		raise KeyError(f"Key '{key}' not found.")

	def clear(self) -> None:
		"""Removes all items from the BD instance, respecting anti-freeze/frozen limits."""
		if self._frozen:
			raise ImmutableError("Cannot clear; the object is frozen.")
		for key in list(self._data.keys()):
			if key not in self._anti_freeze:
				self.delete(key)

	def set(self, path: str, value: Any) -> "BD":
		"""Sets a value at a potentially nested path (e.g., 'user.profile.name')."""
		if not isinstance(path, str):
			raise DataError("Path must be a string.")

		if "." not in path:
			self._set_raw(path, value)
			return self

		parts = path.split(".")
		target = self

		for part in parts[:-1]:
			if part not in target._data or not isinstance(target._data[part], BD):
				target._set_raw(part, BD())
			target = target._data[part]

		target._set_raw(parts[-1], value)
		return self

	def get(self, path: str, default: Any = None) -> Any:
		"""Retrieves a value from a nested path."""
		if "." not in path:
			val = self._data.get(path, KeyError)
			if val is KeyError: return default
			return getattr(self, path) if type(val) is Computed else val

		parts = path.split(".")
		current = self
		for part in parts:
			try:
				current = getattr(current, part) if isinstance(current, (BD, Repr)) else current[part]
			except (AttributeError, KeyError, TypeError):
				return default
		return current

	def delete(self, path: str) -> None:
		"""Deletes a key at a nested path."""
		if "." not in path:
			self._delete_raw(self, path)
			return

		parts = path.split(".")
		target = self
		for part in parts[:-1]:
			target = target._data[part]
		self._delete_raw(target, parts[-1])

	@staticmethod
	def _delete_raw(target: "BD", key: str) -> None:
		if target._frozen and key not in target._anti_freeze:
			raise ImmutableError(f"Cannot delete '{key}'; the object is frozen.")
		if key in target._data:
			del target._data[key]
			target._anti_freeze.discard(key)
		else:
			raise KeyError(f"Key '{key}' not found.")

	def _set_raw(self, key: str, value: Any) -> None:
		if self._frozen and key not in self._anti_freeze:
			raise ImmutableError(f"Cannot modify '{key}'; the object is frozen.")

		old_value = self._data.get(key)

		if type(value) is Validated:
			if not value.validator(value.value):
				raise DataError(f"Validation failed for key '{key}'")
			value = value.value

		if type(value) is Secret:
			pass 
		elif type(value) is AntiFreeze:
			self._anti_freeze.add(key)
			value = value.value

		elif type(value) is Method:
			value = types.MethodType(value.fn, self)
		elif type(value) is Computed and not value.lazy:
			value = value.fn(self)
		elif type(value) is Alias:
			pass
		elif type(value) is Observed:
			callback = value.callback
			value = value.value
			self._watchers.append(lambda k, o, n: callback(o, n) if k == key else None)

		self._data[key] = value

		for watcher in self._watchers:
			try:
				watcher(key, old_value, value)
			except Exception as e:
				warnings.warn(f"Watcher callback failed: {e}")

	def to_dict(self, update_self: bool = True) -> Dict[str, Any]:
		"""Converts the BD structure into a standard Python dictionary."""
		def _serialize(item: Any) -> Any:
			if isinstance(item, Repr):
				return _serialize(item._actual_val)
			if isinstance(item, BD):
				return item.to_dict(update_self=update_self)
			if isinstance(item, (list, tuple, set, frozenset)):
				return [_serialize(i) for i in item]
			if isinstance(item, (dict, MappingProxyType)):
				return {k: _serialize(v) for k, v in item.items()}
			return item

		out = {}
		for k, v in list(self._data.items()):
			if isinstance(v, types.MethodType): continue
			val = getattr(self, k) if type(v) is Computed else v
			serialized_val = _serialize(val)
			out[k] = serialized_val
			if update_self: self._data[k] = serialized_val
		return out

	def freeze(self) -> "BD":
		"""
		Recursively freezes the object, nested BD instances, and converts 
		mutable collections (list, set, dict) into immutable types.
		"""
		super().__setattr__("_frozen", True)

		def _deep_freeze(item: Any) -> Any:
			if isinstance(item, BD):
				return item.freeze()

			if isinstance(item, list):
				return tuple(_deep_freeze(i) for i in item)
			
			if isinstance(item, set):
				return frozenset(_deep_freeze(i) for i in item)

			if isinstance(item, dict):
				return MappingProxyType({k: _deep_freeze(v) for k, v in item.items()})

			return item

		for key, val in list(self._data.items()):
			if key in self._anti_freeze:
				continue

			if not isinstance(val, (types.MethodType, Computed, Alias)):
				self._data[key] = _deep_freeze(val)

		return self

class SlidingTTLCache(Generic[T]):
	"""
	An in-memory cache using a Sliding Window TTL (Time To Live) and LRU eviction.

	Optimized for asyncio by utilizing synchronous dictionary operations which are
	atomic within the context of the single-threaded event loop. Includes an
	optional background garbage collection task to prevent memory bloat.
	"""
	def __init__(self, ttl_seconds: int = 60, maxsize: int = 1000) -> None:
		"""
		Initialize the cache.

		:param ttl_seconds: Seconds of inactivity before an item is considered expired.
		:param maxsize: Maximum number of items to keep in memory.
		"""
		self._ttl = ttl_seconds
		self._maxsize = maxsize
		self._cache: dict[str, list] = {}

		self._gc_thread: threading.Thread | None = None
		self._gc_stop_event: threading.Event | None = None

	def get(self, key: str) -> T | None:
		"""Retrieve an item and refresh its expiration timer."""
		entry = self._cache.pop(key, None)
		if not entry:
			return None

		now = time.monotonic()
		if now >= entry[1]:
			return None

		entry[1] = now + self._ttl
		self._cache[key] = entry
		return entry[0]

	def set(self, key: str, data: T) -> None:
		"""Add an item to cache, enforcing maxsize and TTL."""
		self._cache.pop(key, None)

		if len(self._cache) >= self._maxsize:
			self._cleanup()
			if len(self._cache) >= self._maxsize:
				oldest_key = next(iter(self._cache))
				del self._cache[oldest_key]

		self._cache[key] = [data, time.monotonic() + self._ttl]

	def invalidate(self, key: str) -> None:
		"""Manually remove a specific key from the cache."""
		self._cache.pop(key, None)

	def _cleanup(self) -> None:
		"""
		Internal method to prune all expired keys at once.
		Iterates through all items to find expired ones, as expiration times are not
		guaranteed to be in insertion order due to 'get' refreshing TTL.
		"""
		now = time.monotonic()
		keys_to_remove = [k for k, v in self._cache.items() if now >= v[1]]
		for k in keys_to_remove:
			del self._cache[k]

	def _gc_loop_sync(self, interval: float, stop_event: threading.Event) -> None:
		"""
		The actual loop that sleeps and triggers cleanup in a separate system thread.
		Uses `time.sleep` and a `threading.Event` for control.
		"""
		log.debug(f"SlidingTTLCache _gc_loop_sync started in thread {threading.current_thread().name}")
		try:
			while not stop_event.is_set():
				stop_event.wait(interval)
				if stop_event.is_set():
					break
				self._cleanup()
		except Exception as e:
			log.error("Error in SlidingTTLCache _gc_loop_sync", exc_info=e)
		finally:
			log.debug("SlidingTTLCache _gc_loop_sync stopped.")

	def start_gc(self, interval_seconds: int | None = None) -> None:
		"""
		Starts a background *system thread* to periodically clear expired items.
		"""
		interval = interval_seconds or (self._ttl / 2.0)

		if self._gc_thread is not None and self._gc_thread.is_alive():
			log.warning("SlidingTTLCache background GC thread is already running.")
			return

		self._gc_stop_event = threading.Event()

		self._gc_thread = threading.Thread(
			target=self._gc_loop_sync,
			args=(interval, self._gc_stop_event),
			daemon=True,
			name="SlidingTTLCache_GC_Thread"
		)
		self._gc_thread.start()
		log.info("SlidingTTLCache background GC thread started.")

	def stop_gc(self) -> None:
		"""Cancels the running background garbage collection thread."""
		if self._gc_thread is not None and self._gc_thread.is_alive():
			log.debug("Signaling SlidingTTLCache background GC thread to stop...")
			if self._gc_stop_event:
				self._gc_stop_event.set()
			self._gc_thread.join(timeout=5)
			if self._gc_thread.is_alive():
				log.warning("SlidingTTLCache background GC thread did not terminate gracefully within 5 seconds.")
			self._gc_thread = None
			self._gc_stop_event = None
			log.info("SlidingTTLCache background GC thread stopped.")
		else:
			log.debug("SlidingTTLCache background GC thread is not running.")

class BaseFileIO(ABC, Generic[T]):
	"""
	Abstract Base Class for high-performance, thread-safe File I/O with caching.
	Supports atomic writes and concurrent-safe modifications via context managers.
	"""
	def __init__(self, cache_ttl_seconds: int = 60, cache_maxsize: int = 1000) -> None:
		self.cache = SlidingTTLCache[T](ttl_seconds=cache_ttl_seconds, maxsize=cache_maxsize)
		self._locks: dict[str, RefCountedLock] = {}


	# Internals
	def _acquire_lock(self, path: str) -> RefCountedLock:
		if path not in self._locks:
			self._locks[path] = RefCountedLock()
		lock_obj = self._locks[path]
		lock_obj.count += 1
		return lock_obj

	def _release_lock(self, path: str, lock_obj: RefCountedLock) -> None:
		lock_obj.count -= 1
		if lock_obj.count <= 0:
			self._locks.pop(path, None)

	def _sync_read(self, path: str) -> T:
		with open(path, "rb") as f:
			content = f.read()
		if content.startswith(b'\x1f\x8b'):
			content = gzip.decompress(content)
		return self.deserialize(content)

	def _sync_write(self, path: str, data: bytes, atomic: bool, mode: int | None, verify: bool, compress: bool) -> None:
		if compress:
			data = gzip.compress(data)

		target_path = Path(f"{path}.{os.getpid()}.tmp") if atomic else Path(path)
		final_path = Path(path)

		if final_path.parent:
			final_path.parent.mkdir(parents=True, exist_ok=True)
		
		flags = os.O_WRONLY | os.O_CREAT | os.O_TRUNC
		fd = os.open(str(target_path), flags, mode or 0o666)
		try:
			with os.fdopen(fd, 'wb') as f:
				f.write(data)
				f.flush()
				os.fsync(f.fileno())
			
			if mode is not None:
				target_path.chmod(mode)
				
		except Exception:
			if atomic:
				target_path.unlink(missing_ok=True)
			raise

		if verify:
			with open(target_path, 'rb') as f:
				if f.read() != data:
					if atomic:
						target_path.unlink(missing_ok=True)
					raise IOError("Verification failed: Disk content does not match intended data.")

		if atomic:
			target_path.replace(final_path)

	async def _save_internal(self, path: str, data: T, update_cache: bool, atomic: bool, verify: bool, compress: bool, mode: int | None, skip_if_unchanged: bool) -> bool:
		lock_obj = self._acquire_lock(path)
		try:
			async with lock_obj.lock:
				if skip_if_unchanged:
					cached = self.cache.get(path)
					if cached is not None and cached == data:
						return True

				serialized_data = self.serialize(data)
				await asyncio.to_thread(
					self._sync_write, path, serialized_data, atomic, mode, verify, compress
				)
				if update_cache:
					self.cache.set(path, self.clone(data))
				return True
		except Exception:
			log.error(f"Save failed for {path}", exc_info=True)
			return False
		finally:
			self._release_lock(path, lock_obj)


	# implementation
	@abstractmethod
	def serialize(self, data: T) -> bytes:
		"""Convert data to bytes for disk storage."""
		...

	@abstractmethod
	def deserialize(self, content: bytes) -> T:
		"""Convert bytes from disk back into Python objects."""
		...

	@abstractmethod
	def clone(self, data: T) -> T:
		"""Return a deep copy of the data to prevent accidental mutation of cache."""
		...

	@property
	@abstractmethod
	def default_empty(self) -> T:
		"""The default value to return if a file is missing or corrupted."""
		...


	# Public Methods
	async def load(
		self, 
		path: str | Path, 
		*,
		use_cache: bool = True, 
		read_only: bool = False,
		update_cache: bool = True
	) -> tuple[bool, T]:
		"""
		Load data from cache or disk.

		:param path: The file path to read.
		:param use_cache: If False, ignores existing cache and reads from disk.
		:param read_only: If True, returns the cached object directly (no cloning). 
		                  FASTER, but you MUST NOT modify the returned object.
		:param update_cache: If True, disk-read results will be stored in cache.
		:return: Tuple of (Success Boolean, Data)
		"""
		path = os.path.abspath(os.fspath(path))
		if use_cache:
			cached = self.cache.get(path)
			if cached is not None:
				return True, (cached if read_only else self.clone(cached))

		lock_obj = self._acquire_lock(path)
		try:
			async with lock_obj.lock:
				if use_cache:
					cached = self.cache.get(path)
					if cached is not None:
						return True, (cached if read_only else self.clone(cached))

				try:
					data = await asyncio.to_thread(self._sync_read, path)
					if update_cache:
						self.cache.set(path, self.clone(data))
					return True, data
				except (FileNotFoundError, Exception) as e:
					if not isinstance(e, FileNotFoundError):
						log.error(f"Failed to load {path}", exc_info=True)
					return False, self.default_empty
		finally:
			self._release_lock(path, lock_obj)

	async def save(
		self,
		path: str | Path,
		data: T,
		*,
		update_cache: bool = True,
		atomic: bool = True,
		verify: bool = False,
		compress: bool = False,
		mode: int | None = None,
		background: bool = False,
		skip_if_unchanged: bool = False
	) -> bool:
		"""
		Save data to disk and update cache.

		:param path: The file path to write to.
		:param data: The Python object to save.
		:param update_cache: If True, the data is added to the sliding cache.
		:param atomic: If True, uses a temporary file and os.replace to prevent corruption.
		:param verify: If True, reads the file back to guarantee integrity before returning.
		:param compress: If True, compresses the file with gzip.
		:param mode: Optional file permissions (e.g., 0o600).
		:param background: If True, executes the save as a fire-and-forget asyncio task.
		:param skip_if_unchanged: If True, ignores the save if the data equals the cache.
		:return: True if successful (or successfully dispatched to background), False otherwise.
		"""
		path = os.path.abspath(os.fspath(path))
		if background:
			asyncio.create_task(
				self._save_internal(
					path, self.clone(data), update_cache, atomic, 
					verify, compress, mode, skip_if_unchanged
				)
			)
			return True

		return await self._save_internal(
			path, data, update_cache, atomic, verify, compress, mode, skip_if_unchanged
		)

	@asynccontextmanager
	async def edit(
		self,
		path: str | Path,
		*,
		use_cache: bool = True,
		auto_save: bool = True,
		atomic: bool = True,
		verify: bool = False,
		compress: bool = False,
		mode: int | None = None,
		skip_if_unchanged: bool = False
	) -> AsyncIterator[T]:
		"""
		Context manager to read, modify, and automatically save data.
		The per-file lock is held for the duration of the 'with' block to ensure 
		atomicity and prevent race conditions during the read-modify-write cycle.
		"""
		path = os.path.abspath(os.fspath(path))
		lock_obj = self._acquire_lock(path)
		try:
			async with lock_obj.lock:
				data = None
				original_data = None
				if use_cache:
					cached = self.cache.get(path)
					if cached is not None:
						data = self.clone(cached)

				if not data:
					try:
						data = await asyncio.to_thread(self._sync_read, path)
					except FileNotFoundError:
						data = self.clone(self.default_empty)
					except Exception as e:
						log.error(f"Critical error reading {path} during edit. Falling back to default.", exc_info=e)
						raise

				if auto_save and skip_if_unchanged:
					original_data = self.clone(data)

				yield data

				if auto_save:
					if skip_if_unchanged and original_data == data:
						return

					serialized = self.serialize(data)
					await asyncio.to_thread(
						self._sync_write, path, serialized, atomic, mode, verify, compress
					)
					self.cache.set(path, self.clone(data))
		finally:
			self._release_lock(path, lock_obj)

			response.raise_for_status()

				tmp.write(response.content)
				tmp_path = tmp.name

		log.debug(f"Temporary file created at: {tmp_path}")
		try:
			return await self._upload_and_wait(tmp_path)
		finally:
			tmp_file = Path(tmp_path)
			if tmp_file.exists():
				tmp_file.unlink()
				log.debug(f"Cleaned up temporary file: {tmp_path}")

				return await self._handle_remote_file(item)
			if Path(item).is_file():
				log.debug(f"Detected local file path: {item}")
				return await self._upload_and_wait(item)

		if isinstance(item, bytes):
			log.debug("Detected raw bytes input")
			return {"mime_type": "application/octet-stream", "data": item}

		return item

				processed_parts = await asyncio.gather(
					*[self._process_input(i) for i in inputs]
				)

			log.debug("Sending payload to Gemini API")
			response = await self.model.generate_content_async(payload)

			if response.candidates and len(response.candidates) > 0:
				log.info("Content generation successful")
				return True, response.text

			log.warning("AI returned no candidates or content")
			return False, "AI returned no content."

		except Exception as e:
			log.error("Error during call execution", exc_info=True)
			return False, f"Utility Error: {str(e)}"

			perms.administrator
			if perms else False
		)
		return is_owner or is_dev or is_admin
	return commands.check(predicate)

def create_id() -> str:
	BASE62_CHARS = string.digits + string.ascii_lowercase + string.ascii_uppercase

	def base62_encode(number, length):
		encoded_string = ""
		base = len(BASE62_CHARS)
		while number > 0:
			remainder = number % base
			number //= base
			encoded_string = BASE62_CHARS[remainder] + encoded_string
		return encoded_string.zfill(length)

	timestamp_s = int(time.time())

	random_part_max = 62 * 62 - 1
	random_value = random.randint(0, random_part_max)

	combined_int = (timestamp_s * (62 * 62)) + random_value
	id_str = base62_encode(combined_int, 10)
	return id_str

def em(text: str) -> str:
	MARKDOWN_CHARS = r"\*_~`|>#-.()[]{}!"
	result = []
	for ch in text:
		if ch in MARKDOWN_CHARS:
			result.append("\\" + ch)
		else:
			result.append(ch)
	return "".join(result)

def mt(*args: str | int, sep: str = "\n") -> str:
	joined = sep.join(map(str, args))
	return joined

try:
	DISCORD_MODELS = (
		discord.Member, discord.Role, discord.Emoji, discord.Guild,
		discord.TextChannel, discord.VoiceChannel, discord.CategoryChannel,
		discord.User, discord.Message, discord.Asset, discord.Client,

	)

except ImportError:
	DISCORD_MODELS = ()
	DISCORD_TYPES = ()


class Convert:
	def __init__(
		self, 
		filter_attrs: list = None, 
		max_depth: int = 5,
		ignore_none: bool = True,
		ignore_empty: bool = True,
		flatten: bool = False,
		output_format: str = "dict"
	):
		self.filter_attrs = set(filter_attrs or [])
		self.max_depth = max_depth
		self.ignore_none = ignore_none
		self.ignore_empty = ignore_empty
		self.flatten = flatten
		self.output_format = output_format

	def _flatten_dict(self, d: dict, parent_key: str = '', sep: str = '_') -> dict:
		items = []
		for k, v in d.items():
			new_key = f"{parent_key}{sep}{k}" if parent_key else str(k)
			if isinstance(v, dict):
				items.extend(self._flatten_dict(v, new_key, sep=sep).items())
			else:
				items.append((new_key, v))
		return dict(items)

	def _serialize(self, obj, seen_stack, depth):
		if self.ignore_none and obj is None:
			return None

		if isinstance(obj, (int, float, bool, str)):
			if self.ignore_empty and obj == "":
				return None
			return obj

		if isinstance(obj, datetime.datetime):
			return obj.isoformat()

		if DISCORD_TYPES and isinstance(obj, DISCORD_TYPES):
			return str(obj)

		if hasattr(obj, 'url') and isinstance(getattr(obj, 'url'), str):
			return getattr(obj, 'url')

		if depth >= self.max_depth:
			return str(obj) if not callable(obj) else None

		obj_id = id(obj)
		if obj_id in seen_stack:
			return "<Circular>"

		seen_stack.add(obj_id)
		result = None

		try:
			if isinstance(obj, dict):
				result = {}
				for k, v in obj.items():
					val = self._serialize(v, seen_stack, depth + 1)
					if self.ignore_none and val is None: continue
					if self.ignore_empty and val in ({}, []): continue
					result[str(k)] = val

			elif not (DISCORD_MODELS and isinstance(obj, DISCORD_MODELS)):
				try:
					result = []
					for item in obj:
						val = self._serialize(item, seen_stack, depth + 1)
						if self.ignore_none and val is None: continue
						if self.ignore_empty and val in ({}, []): continue
						result.append(val)
				except TypeError:
					pass

			if result is None:
				result = {}
				for attr_name in dir(obj):
					if attr_name.startswith('_') or attr_name in self.filter_attrs:
						continue

					if depth > 1 and attr_name in ('members', 'roles', 'channels', 'guilds'):
						continue

					try:
						value = getattr(obj, attr_name)
						if callable(value):
							continue

						serialized_val = self._serialize(value, seen_stack, depth + 1)

						if self.ignore_none and serialized_val is None:
							continue
						if self.ignore_empty and serialized_val in ({}, [], ""):
							continue

						result[attr_name] = serialized_val
					except Exception:
						pass
		finally:
			seen_stack.discard(obj_id)

		return result

	async def __call__(self, obj, **kwargs):
		original_state = {k: getattr(self, k) for k in kwargs if hasattr(self, k)}
		for k, v in kwargs.items():
			if hasattr(self, k):
				setattr(self, k, v)

		def process():
			raw_dict = self._serialize(obj, set(), 0)

			if self.flatten and isinstance(raw_dict, dict):
				raw_dict = self._flatten_dict(raw_dict)

			if self.output_format == "yaml":
				return yaml.dump(raw_dict, default_flow_style=False, sort_keys=False)
			elif self.output_format == "json":
				return json.dumps(raw_dict)

			return raw_dict

		result = await asyncio.to_thread(process)

		for k, v in original_state.items():
			setattr(self, k, v)

		return result

def extract(text: str) -> BD:
	log.debug("Extracting code and monologue from text")
	py = re.search(
		r"<code>\s*(.*?)\s*</code>",
		text, re.DOTALL
	)
	mo = re.search(
		r"<monologue>\s*(.*?)\s*</monologue>",
		text, re.DOTALL
	)
	log.debug(f"Extracted Code:\n{py}\n\nExtracted Monologue:\n{mo}")

	return BD(
		code = py.group(1).strip() if py else "",
		monologue = mo.group(1).strip() if mo else ""
	)

class StreamInterceptor(io.StringIO):
	def __init__(self, callback: Optional[Callable[[str], None]] = None):
		super().__init__()
		self._callback = callback

	def write(self, s: str) -> int:
		res = super().write(s)
		if self._callback:
			self._callback(s)
		return res

class Code:
	tasks = BD()

	def __init__(self, code_str: str, on_stdout: Optional[Callable[[str], None]] = None, on_stderr: Optional[Callable[[str], None]] = None):
		self.code_str = code_str
		self.execution_id = f"EI{str(create_id())}"
		self.data = BD(
			status = status.PENDING,
			stdout = "",
			stderr  = "",
			result = None,
			error = None
		)
		self._task: Optional[asyncio.Task] = None

		self._stdout_buf = StreamInterceptor(on_stdout)
		self._stderr_buf = StreamInterceptor(on_stderr)

		Code.tasks.set(self.execution_id, self)
		log.debug(f"Code task initialized: {self.execution_id}")

	@staticmethod
	def get_task(execution_id: str) -> Optional['Code']:
		return getattr(Code.tasks, execution_id, None)

	async def execute(self, timeout: Optional[float] = None, **context_vars: Any) -> asyncio.Task:
		if self._task and not self._task.done():
			log.warning(f"Execution attempt on already running task: {self.execution_id}")
			raise RuntimeError("This code block is already running. Call stop() first.")

		self._reset_state()
		self.data.status = status.RUNNING
		log.info(f"Starting execution for ID: {self.execution_id}")

		user_module = types.ModuleType(self.execution_id)
		safe_builtins = dict(config.sandbox.allowed_builtins)
		setattr(user_module, "__builtins__", safe_builtins)

		try:
			exec(self.code_str, user_module.__dict__)
			main_func = getattr(user_module, "main", None)

			if not main_func:
				raise ValueError("User code must define an 'async def main(**kwargs)' function.")
			if not inspect.iscoroutinefunction(main_func):
				raise ValueError("'main' must be an async function.")

		except Exception as e:
			log.error(f"Setup error for ID {self.execution_id}: {str(e)}")
			self.data.error = f"Compilation/Setup Error:\n{traceback.format_exc()}"
			self.data.status = status.FAILED
			raise

		async def task_wrapper():
			with redirect_stdout(self._stdout_buf), redirect_stderr(self._stderr_buf):
				try:
					if timeout:
						log.debug(f"Executing ID {self.execution_id} with {timeout}s timeout")
						self.data.result = await asyncio.wait_for(
							main_func(**context_vars), 
							timeout=timeout
						)
					else:
						self.data.result = await main_func(**context_vars)

					self.data.status = status.COMPLETED
					log.info(f"Execution completed for ID: {self.execution_id}")

				except asyncio.TimeoutError:
					log.warning(f"Execution timed out for ID: {self.execution_id}")
					self.data.error = f"Execution timed out after {timeout} seconds."
					self.data.status = status.TIMED_OUT

				except asyncio.CancelledError:
					log.info(f"Execution cancelled for ID: {self.execution_id}")
					self.data.status = status.CANCELLED

				except Exception as e:
					err_msg = traceback.format_exc()
					log.error(f"Runtime error for ID {self.execution_id}:\n{err_msg}")
					self.data.error = str(e)
					self.data.status = status.FAILED

				finally:
					self._sync_data()

		self._task = asyncio.create_task(task_wrapper())
		return self._task

	async def stop(self) -> tuple[bool, str]:
		if not self._task:
			log.debug(f"Stop called on null task: {self.execution_id}")
			return False, "No active task to stop."
		if self._task.done():
			log.debug(f"Stop called on finished task: {self.execution_id}")
			return False, f"Task already finished with status: {self.data.status}."

		log.info(f"Cancelling task ID: {self.execution_id}")
		self._task.cancel()
		try:
			await self._task
		except asyncio.CancelledError:
			pass

		return True, "Task stopped successfully."

	async def wait(self) -> BD:
		if self._task and not self._task.done():
			log.debug(f"Waiting for task ID: {self.execution_id}")
			await self._task
		self._sync_data()
		return self.data

	def get_output(self) -> BD:
		self._sync_data()
		return self.data

	def _sync_data(self):
		self.data.stdout = self._stdout_buf.getvalue()
		self.data.stderr = self._stderr_buf.getvalue()
		log.debug(f"Synced buffers for ID: {self.execution_id} (stdout: {len(self.data.stdout)} bytes)")

	def _reset_state(self):
		log.debug(f"Resetting state for ID: {self.execution_id}")
		self.data = BD(
			status = status.PENDING,
			stdout = "",
			stderr  = "",
			result = None,
			error = None
		)
		self._stdout_buf.seek(0)
		self._stdout_buf.truncate(0)
		self._stderr_buf.seek(0)
		self._stderr_buf.truncate(0)

class Paginated(ui.LayoutView):
	def __init__(self, header: str, items: list[str | list[str]], user: discord.Member | discord.User, accent_color: int = 0xb3e7df, per_page: int = 5, char_limit: int = 1800) -> None:
		super().__init__()
		self.user = user
		self.header = header
		self.raw_items = items
		self.accent_color = accent_color
		self.per_page = per_page
		self.char_limit = char_limit
		self.current_page = 0

		self.pages = self._calculate_pages()
		self.total_pages = len(self.pages)

		self.create_layout()

	def _calculate_pages(self) -> list:
		pages = []
		current_chunk = []
		current_length = len(self.header)

		remaining_items = list(self.raw_items)

		while remaining_items:
			item = remaining_items.pop(0)

			text_content = item[0] if isinstance(item, list) else item
			item_len = len(text_content)

			if item_len > self.char_limit:
				trunc_limit = self.char_limit - 30
				cut_pos = text_content.rfind(" ", 0, trunc_limit)
				if cut_pos == -1: cut_pos = trunc_limit

				part_one = text_content[:cut_pos] + "... goto next page."
				part_two = text_content[cut_pos:].strip()

				if isinstance(item, list):
					remaining_items.insert(0, [part_two, item[1]])
					item = [part_one, item[1]]
				else:
					remaining_items.insert(0, part_two)
					item = part_one

				item_len = len(part_one)

			if len(current_chunk) >= self.per_page or (current_length + item_len) > self.char_limit:
				if current_chunk:
					pages.append(current_chunk)
					current_chunk = []
					current_length = len(self.header)

			current_chunk.append(item)
			current_length += item_len

		if current_chunk:
			pages.append(current_chunk)
		
		return pages or [[]]

async def resolve_target(ctx: commands.Context, scope: str) -> Tuple[Any, Optional[int]]:
	if scope not in ('server', 'user'):
		await ctx.reply(view=Fail(
			"Invalid Scope",
			f"Scope must be either 'server' or 'user', not '{scope}'."
		))
		return None, None

	if scope == 'server':
		if not ctx.guild:
			await ctx.reply(view=Fail(
				"Invalid Scope",
				"This command must be used in a server if the scope passed is 'server'."
			))
			return None, None
		return server, ctx.guild.id

	return user, ctx.author.id

	success, data = await manager.settings.load(target_id)

		description='Group for prefix commands.'
	)
	async def prefix(self, ctx: commands.Context):
		"""Group for prefix commands."""
		if ctx.invoked_subcommand is not None:
			return

		await ctx.reply(view=Fail(
			"Subcommand Missing",
			"Available subcommands:",
			"```",
			"add    (a)",
			"remove (r)",
			"list   (l)",

		))


	@prefix.command(
		name='add',
		aliases=['a'],
		description='Add a prefix to your personal profile or current server.'
	)
	@app.describe(
		type='Whether to add the prefix to your personal profile or current server',
		char='The character to add as prefix'
	)
	@is_admin(config.dev_ids, ('type', 'user'))
	async def add(self, ctx: commands.Context, type: Literal['server', 'user'] = 'user', *, char: str):
		"""
		Add a prefix to your personal profile or current server, allows you to run commands by your own prefix
		Example: /setting prefix add user !
		"""
		if len(char) > 2:
			await ctx.reply(view=Fail(
				"Prefix Too Long",
				"Prefix must be under 2 characters long."
			))
			return

		manager, target_id = await resolve_target(ctx, type)
		if not manager:
			return

		data = await load_settings(manager, target_id, type)

		if len(data["prefix"]) >= config.max_prefix:
			await ctx.reply(view=Fail(
				"Max Prefix Limit Reached",
				f"You can only add {config.max_prefix} prefixes.{f'.. Well the current amount is passed that but you still cannot add more.' if len(data['prefix']) > config.max_prefix else ''}"
			))
			return
		if char in data["prefix"]:
			await ctx.reply(view=Fail(
				"Prefix Already exists",
				f"Prefix '{em(char)}' already in the settings, you don't need to add it twice."
			))
			return
		if char in config.prefix:
			await ctx.reply(view=Fail(
				"Prefix Reserved By Bot",
				f"Prefix '{em(char)}' is reserved by the bot and used globaly, you don't need to add it to use it."
			))

		self.bot = bot


	@dev.command(
		name='code',
		aliases=['c'],
		description='Execute python code'
	)
	@app.describe(code='The code to execute')
	async def code(self, ctx: commands.Context, *, code: str):
		"""
		Execute python code. you are required to define an asynchronios main function as entry point, accepting message and bot object.
		Example: /dev execute async def main(message, bot): await message.channel.send("Hello there.")
		"""
		await ctx.defer(ephemeral=False)

		runner = Code(code)

		try:
			await runner.execute(timeout=30.0, message=ctx.message, bot=self.bot)

			data = await runner.wait()

			output_parts = []

			if data.stdout:
				output_parts.append(f"**Stdout:**\n```py\n{data.stdout}\n```")
			if data.stderr:
				output_parts.append(f"**Stderr:**\n```py\n{data.stderr}\n```")
			if data.error:
				output_parts.append(f"**Error:**\n```py\n{data.error}\n```")
			if data.result is not None:
				output_parts.append(f"**Result:** `{data.result}`")

			final_content = "\n".join(output_parts) if output_parts else "Code executed with no output."

			if data.status == status.COMPLETED:
				await ctx.reply(view=Success(f"**Execution {runner.execution_id} Successful**\n\n{final_content}"))
			else:
				await ctx.reply(view=Fail(f"**Execution {runner.execution_id} {data.status}**\n\n{final_content}"))

		except Exception as e:
			await ctx.reply(view=Fail(f"An unexpected error occurred during execution setup: {str(e)}"))


	@dev.command(
		name='reload_config',
		aliases=['r'],
		description='Reload the config file'
	)

		original_config = config.to_dict()

	async def autocomplete(interaction: discord.Interaction, current: str):
		all_extensions = await extension.get_all()
		choices = []

		loaded_exts = [ext.lower() for ext in interaction.bot.extensions.keys()]

		for ext_path in all_extensions:
			if current.lower() in ext_path.lower():
				prefix = "[+]" if ext_path.lower() in loaded_exts else "[-]"

class CreatePath:
	async def process_path(self, path: str | Path, is_dir: bool = False) -> None:
		path_obj = Path(path)

		if is_dir:
			dir_to_create = path_obj
		else:
			dir_to_create = path_obj.parent

		try:
			if dir_to_create and not dir_to_create.exists():
				log.debug(f"Creating directory: {dir_to_create}")
				await asyncio.to_thread(dir_to_create.mkdir, parents=True, exist_ok=True)

			if not is_dir:
				log.debug(f"Touching file: {path_obj}")
				await asyncio.to_thread(path_obj.touch, exist_ok=True)
		except Exception:
			log.error(f"An unexpected error occurred while processing {path}", exc_info=True)

	async def __call__(self, *paths: tuple[str | Path, bool] | str | Path) -> None:
		log.debug(f"Starting path processing for {len(paths)} paths.")
		tasks = []
		for path in paths:
			if isinstance(path, tuple):
				tasks.append(self.process_path(*path))
			else:
				tasks.append(self.process_path(path))
		await asyncio.gather(*tasks)
		log.debug("Path processing batch complete.")
create_path = CreatePath()

def setup_directory() -> None:
	log.info("Initializing filesystem setup...")

	asyncio.run(create_path(
		(base.data,  	True),
		(base.logs,  	True),
		(sub.servers,	True),
		(sub.users,  	True),
		(sub.trash,  	True),
		(sub.glob,   	True),

		file.glob.blacklist,
		file.glob.bot_activity
	))

	log.info("Filesystem setup complete.")

for obj in (base, sub, file): obj.freeze()
